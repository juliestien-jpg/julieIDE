# Contre-recette M5 : titres des volets de la fusion pendant un cherry-pick en conflit (sans mise de côté : on passe par un a.txt propre).
# Décor : contre-recette.sh <dossier> pick, mais a.txt est remis propre par le scénario : voir contre-m5-run.sh
project @A@
resize 1280 720
wait-idle
ui git.tab log
ui git.log-filter branch feature/login
wait-idle
wait 500
ui menu.open log 2
ui menu.choose log.cherry_pick
wait-idle
wait 1200
ui dialog.choose stash
wait-idle
wait 1500
ui git.tab changes
state @OUT@/m5-pick-etat.json
ui merge.open a.txt
wait-idle
wait 800
state @OUT@/m5-pick-merge.json
screenshot @OUT@/m5-pick-merge.png
ui merge.close
wait 300
quit
